# Accepted lint findings, read by run-lint.sh.
#
# One waiver per line:  <phase> <regex>
#   <phase>  tags | verilator | yosys | reset | slang | verible
#   <regex>  extended regex (grep -E), matched against one finding line as it
#            appears in <phase>.all of a run-lint.sh work directory
# A finding that matches is waived. Put the reason on a comment line above it.
#
# Guideline: resolve a finding at its root; waive only as the last resort, by
# the same steps as openPCIE's run-lint-waivers.txt:
#   1. dead logic in our RTL: delete it;
#   2. oversized buses in our RTL: size them to what is used;
#   3. unused outputs of hard macros, primitives and third-party cores:
#      .PIN (/*LINT:OK*/), or route them to an NC_ (no-connect) signal;
#   4. only then a waiver here, with its reason.

# NC_ signals are no-connects by name: driven by an unused output, never read
verilator ^%Warning-UNUSEDSIGNAL: .*'NC_[A-Za-z0-9_]+'

# The SOC CPU address (soc_addr_t): its nc_ fields are not decoded
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/7\.soc/soc_top\.sv:[0-9]+:[0-9]+: Bits of signal are not used: 'addr'\[31:29,27:13,1:0\]$

# dll_if: an RX TLP beat carries no half word (rx_tlp.half is 0): the TL
# does not read it
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/4\.dll/dll_if\.sv:[0-9]+:[0-9]+: Bits of signal are not used: 'rx_tlp'\[64\]$

# KEEPME: the SOC's rvfi (the CPU's retired instructions) is read only by
# WaveCrux's RISC-V views (tb_soc, 5.sim) and formal checks, never by the RTL
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/7\.soc/rvfi_if\.sv:[0-9]+:[0-9]+: Signal is not used: 'rvfi_[a-z0-9_]+'$

# PhyStatus is a PIPE signal the PIPE drives, but Cologne Chip's MAC does not
# read it (no wait for the PHY after a power-state change or receiver detect),
# issue #12
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/2\.pipe/pipe_if\.sv:[0-9]+:[0-9]+: Signal is not used: 'phy_status'$

# The SerDes BFM is a behavioural model, not RTL: its # delays are computed
verilator ^%Warning-ZERODLY: 2\.rtl/1\.phy/2\.openSERDES_BFM/1\.gatemate/1\.model/

# Generated by PeakRDL: left as the generator writes it (as on openPCIE)
verilator ^%Warning-UNUSEDSIGNAL: csr/csr\.sv:[0-9]+:[0-9]+: Signal is not used: '(decoded_err|is_valid_addr|is_valid_rw)'
verilator ^%Warning-MULTIDRIVEN: csr/csr\.sv:[0-9]+:[0-9]+: Variable written to in always_(comb|ff) also written by other process .*: '(field_combo|field_storage)'$

# Descriptor names: a signal or function argument of a tl_pkg header or
# sideband type (*_dw0 ... *_dw3, *_hdr_*, *_tuser) reads only the fields it
# needs
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/(5\.tlp|6\.trn_bridge)/[a-z_]+\.sv:[0-9]+:[0-9]+: Bits of (signal|function variable) are not used: '[a-z0-9_]*(_dw[0-9]+|_hdr_[a-z0-9_]+|_tuser)'

# AXI-Stream fields kept to the standard (as on openPCIE): the SOC does not
# read the RX tkeep and tuser (rx_tuser_t), trn_rx drives them constant; on
# TX the TL takes the length from the header and supports no qualifier
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/6\.trn_bridge/axis_if\.sv:[0-9]+:[0-9]+: (Signal is not used: 'tkeep'|Bits of signal are not used: 'tkeep'\[6:0\]|Signal is not used: 'tuser')$

# trn_compat keeps the UG477 TX inputs this core does not support, so user
# logic written for TRN connects unchanged
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/5\.tlp/trn_compat\.sv:[0-9]+:[0-9]+: Signal is not used: 'trn_(trem|tsrc_dsc|terrfwd|tstr|tecrc_gen)'$

# The TX buffer is read one word at a time: the look-ahead word of tlp_queue
# (rd.next) is for tlp_rx_dispatch
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/5\.tlp/tl_top\.sv:[0-9]+:[0-9]+: Bits of signal are not used: 'app_q_rd'\[63:0\]$

# Wire-format structs read field by field: the framing symbols (STP, SDP,
# END) were already matched by packet_handler; a TS's N_FTS and data rate are
# not read (Gen1 only, no L0s)
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/4\.dll/packet_handler\.sv:[0-9]+:[0-9]+: Bits of signal are not used: 'prefix'\[7:0\]$
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/4\.dll/virtual_channel_rx\.sv:[0-9]+:[0-9]+: Bits of signal are not used: 'dllp_data'\[63:56,7:0\]$
verilator ^%Warning-UNUSEDSIGNAL: 2\.rtl/3\.mac/ccfpga_OS_process\.sv:[0-9]+:[0-9]+: Bits of signal are not used: 'i_data'\[39:24\]$

# every unique case has a default, also a full one (our rule): unique checks
# that the items never overlap, the default that no value goes unhandled
slang ^2\.rtl/.*\.sv:[0-9]+:[0-9]+: warning: .case[zx]?. marked .(unique|priority). has .default. label

# reset: data and address flops need no reset (our rule: control flops and
# flags do). These reach a select through a compare or an address decode, each
# anchored to its instance path: eduBOS5's operands, load/store offset, data
# address, ALU result and operand signs; the instruction word from the RAM
reset 'u_pcie_top\.u_soc\.cpu\.(rs1_reg|ls_offset_reg|dmem_addr|u_alu\.(alu_minus_reg|aluin_[12]_31_reg))'
reset 'u_pcie_top\.u_soc\.imem_rdat'
